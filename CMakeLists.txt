cmake_minimum_required(VERSION 3.25)
project(cwtrainerzero VERSION 0.3.0 LANGUAGES C CXX)
include(CTest)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(CW_BUILD_UI "Build the LVGL app" ON)
option(CW_CARDPUTER_ZERO "Build for Cardputer Zero Linux aarch64 (not the ESP32 Cardputer)" OFF)
set(CW_LVGL_SOURCE "" CACHE PATH "Optional pre-fetched LVGL v9.5.0")
set(CW_MINIAUDIO_SOURCE "" CACHE PATH "Optional pre-fetched miniaudio 0.11.25")

add_library(cw-core STATIC src/trainer.cpp)
target_include_directories(cw-core PUBLIC src)
target_compile_options(cw-core PRIVATE -Wall -Wextra -Wpedantic)
if(BUILD_TESTING AND NOT CMAKE_CROSSCOMPILING)
  add_executable(cw-tests tests/test_trainer.cpp)
  target_link_libraries(cw-tests PRIVATE cw-core)
  target_compile_options(cw-tests PRIVATE -Wall -Wextra -Wpedantic)
  add_test(NAME cw-core COMMAND cw-tests)
endif()

if(CW_BUILD_UI)
  include(FetchContent)
  set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
  set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
  set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/src/lv_conf.h" CACHE FILEPATH "" FORCE)
  if(CW_LVGL_SOURCE)
    set(FETCHCONTENT_SOURCE_DIR_LVGL "${CW_LVGL_SOURCE}")
  endif()
  if(CW_MINIAUDIO_SOURCE)
    set(FETCHCONTENT_SOURCE_DIR_MINIAUDIO "${CW_MINIAUDIO_SOURCE}")
  endif()
  FetchContent_Declare(lvgl GIT_REPOSITORY https://github.com/lvgl/lvgl.git
    GIT_TAG 85aa60d18b3d5e5588d7b247abf90198f07c8a63)
  FetchContent_Declare(miniaudio GIT_REPOSITORY https://github.com/mackron/miniaudio.git
    GIT_TAG 9634bedb5b5a2ca38c1ee7108a9358a4e233f14d SOURCE_SUBDIR cmake-not-needed)
  FetchContent_MakeAvailable(lvgl miniaudio)
  target_compile_definitions(lvgl PUBLIC CW_CARDPUTER_ZERO=$<BOOL:${CW_CARDPUTER_ZERO}>)

  find_package(Threads REQUIRED)
  add_executable(cwtrainerzero src/main.cpp src/audio.cpp)
  target_include_directories(cwtrainerzero PRIVATE "${miniaudio_SOURCE_DIR}")
  target_compile_options(cwtrainerzero PRIVATE -Wall -Wextra -Wpedantic)
  target_link_libraries(cwtrainerzero PRIVATE cw-core lvgl::lvgl Threads::Threads ${CMAKE_DL_LIBS} m)
  if(CW_CARDPUTER_ZERO)
    if(NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64)$")
      message(FATAL_ERROR "CW_CARDPUTER_ZERO requires an ARM64 Linux target or the official BSP cross-toolchain.")
    endif()
    include(cmake/cardputer-package.cmake)
  else()
    find_package(SDL2 CONFIG REQUIRED)
    target_link_libraries(cwtrainerzero PRIVATE SDL2::SDL2)
    if(BUILD_TESTING)
      add_test(NAME cw-simulator-workflow COMMAND cwtrainerzero --smoke-test "${CMAKE_BINARY_DIR}/smoke")
      set_tests_properties(cw-simulator-workflow PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy" TIMEOUT 75)
    endif()
  endif()
endif()
